import { readFileSync, writeFileSync } from 'fs';
import { OpenAI } from 'openai';
import dotenv from 'dotenv';

dotenv.config();

const openai = new OpenAI({
  apiKey: process.env.OPENROUTER_API_KEY,
  baseURL: "https://openrouter.ai/api/v1"
});

const MODEL = "anthropic/claude-3.5-sonnet";   // Change if you prefer another model

async function callLLM(prompt, systemMessage = "You are a helpful assistant.") {
  const response = await openai.chat.completions.create({
    model: MODEL,
    messages: [
      { role: "system", content: systemMessage },
      { role: "user", content: prompt }
    ],
    temperature: 0.3,
    max_tokens: 16000
  });
  return response.choices[0].message.content.trim();
}

async function improveMoviePage(movieFileName) {
  try {
    console.log(`🚀 Starting improvements for: ${movieFileName}`);

    // Read original HTML
    const originalHtml = readFileSync(movieFileName, 'utf-8');

    // Read the improvement prompt from file
    const basePrompt = readFileSync('improvement-prompt.txt', 'utf-8');

    // Step 1: LLM1 - Initial Improvement
    console.log("📝 Pass 1: Initial improvement...");
    const llm1Prompt = `${basePrompt}\n\nHere is the original HTML:\n\n\`\`\`html\n${originalHtml}\n\`\`\``;
    const llm1Output = await callLLM(llm1Prompt, "You are an expert SEO and web developer.");

    // Step 2: LLM2 - Refinement
    console.log("🔄 Pass 2: Refinement...");
    const llm2Prompt = `Here is the first improved version:\n\n\`\`\`html\n${llm1Output}\n\`\`\`\n\nPlease further refine and polish this HTML. Make it even better while following the original guidelines.`;
    const llm2Output = await callLLM(llm2Prompt, "You are an expert editor and SEO specialist.");

    // Step 3: LLM Review - Synthesize and Finalize
    console.log("✅ Pass 3: Review and synthesis...");
    const reviewPrompt = `You are an expert reviewer. Here are two improved versions of the same book-vs-movie comparison page:

Version 1 (LLM1):
\`\`\`html
${llm1Output}
\`\`\`

Version 2 (LLM2):
\`\`\`html
${llm2Output}
\`\`\`

Synthesize the best elements from both versions into one final, superior HTML page. Choose the strongest improvements from each while maintaining consistency, quality, and adherence to all guidelines. Return ONLY the final clean HTML.`;
    
    const finalHtml = await callLLM(reviewPrompt, "You are a meticulous final editor.");

    // Save the final improved file
    const outputFileName = movieFileName.replace('.html', '-improved-final.html');
    writeFileSync(outputFileName, finalHtml, 'utf-8');

    console.log(`🎉 Success! Final improved file saved as: ${outputFileName}\n`);

  } catch (error) {
    console.error(`❌ Error processing ${movieFileName}:`, error.message);
  }
}

// ====================== RUN ======================
const movieFile = process.argv[2];

if (!movieFile) {
  console.error("Usage: node film-improvements.js <movie-filename.html>");
  console.error("Example: node film-improvements.js animal-farm.html");
  process.exit(1);
}

improveMoviePage(movieFile);
