  <script>
    function processLinks() {
      const tag = document.getElementById('tag').value.trim();
      const input = document.getElementById('inputList').value.trim();
      const logEl = document.getElementById('log');
      const tbody = document.querySelector('#resultTable tbody');
      
      if (!input) return;

      const lines = input.split('\n').filter(l => l.trim());
      tbody.innerHTML = '';
      logEl.innerHTML = '';
      
      let successCount = 0;
      let failCount = 0;

      lines.forEach((line, index) => {
        const time = new Date().toLocaleTimeString([], { hour12: false });
        
        // Extract ASIN (10 alphanumeric chars) 
        let asinMatch = line.match(/(?:\/dp\/|ASIN=|asin:)([A-Z0-9]{10})/i);
        let asin = asinMatch ? asinMatch[1].toUpperCase() : (line.trim().length === 10 ? line.trim().toUpperCase() : null);

        const logEntry = document.createElement('div');
        logEntry.className = 'log-entry';

        if (asin && /^[A-Z0-9]{10}$/.test(asin)) {
          const url = `https://www.amazon.com/dp/${asin}?tag=${tag}`;
          const html = `<a class="book-affiliate" href="${url}" target="_blank" rel="noopener">Buy on Amazon →</a>`;

          // Update Table
          const row = tbody.insertRow();
          row.innerHTML = `<td>${asin}</td><td><a href="${url}" target="_blank">Link</a></td><td><code>${html.replace(/</g, '&lt;')}</code></td>`;
          
          // Update Log
          logEntry.innerHTML = `[${time}] <span class="success">SUCCESS:</span> Processed ASIN ${asin}`;
          successCount++;
        } else {
          logEntry.innerHTML = `[${time}] <span class="failure">FAILURE:</span> Could not resolve ASIN for line ${index + 1}: "${line.substring(0,20)}..."`;
          failCount++;
        }
        
        logEl.appendChild(logEntry);
      });

      // Final Summary Log
      const summary = document.createElement('div');
      summary.className = 'stats';
      summary.innerHTML = `--- FINAL REPORT ---<br>Total Lines: ${lines.length}<br>Success: ${successCount}<br>Failures: ${failCount}`;
      logEl.appendChild(summary);
    }
  </script>
